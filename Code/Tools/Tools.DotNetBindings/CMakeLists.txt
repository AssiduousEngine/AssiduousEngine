# Tools.DotNetBindings - writes the engine's C# bindings, generated from its reflection.
add_executable(Tools.DotNetBindings Main.cpp)
# The whole script surface: every facade and every component's reflection, as a game's scripts see it.
target_link_libraries(Tools.DotNetBindings PRIVATE Foundation::Script.DotNet Engine::Composition)

# The engine's managed runtime (Engine.Runtime.dll and its runtimeconfig), built with the bindings
# into DotNet/ beside the binaries, where the engine looks for it.
find_program(DOTNET_EXECUTABLE dotnet REQUIRED)
# Global (cached), since dotnet_link_compiled_runtime below is called from other folders.
set(DOTNET_RUNTIME_SOURCE ${CMAKE_SOURCE_DIR}/Code/Foundation/Script.DotNet/Managed CACHE INTERNAL "")
set(DOTNET_RUNTIME_OUT ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/DotNet CACHE INTERNAL "")
set(DOTNET_RUNTIME_BINDINGS ${CMAKE_CURRENT_BINARY_DIR}/Engine.g.cs CACHE INTERNAL "")
add_custom_command(
    OUTPUT ${DOTNET_RUNTIME_BINDINGS}
    COMMAND Tools.DotNetBindings ${DOTNET_RUNTIME_BINDINGS}
    DEPENDS Tools.DotNetBindings
    COMMENT "Generating the C# bindings")
add_custom_command(
    OUTPUT ${DOTNET_RUNTIME_OUT}/Engine.Runtime.dll ${DOTNET_RUNTIME_OUT}/Engine.Runtime.runtimeconfig.json
    COMMAND ${DOTNET_EXECUTABLE} build ${DOTNET_RUNTIME_SOURCE}/Engine.Runtime.csproj -c Debug -o ${DOTNET_RUNTIME_OUT}
            -p:BaseIntermediateOutputPath=${CMAKE_CURRENT_BINARY_DIR}/RuntimeObj/
            -p:EngineBindings=${DOTNET_RUNTIME_BINDINGS} -nologo -v:q
    DEPENDS ${DOTNET_RUNTIME_SOURCE}/Engine.Runtime.csproj ${DOTNET_RUNTIME_SOURCE}/Bridge.cs ${DOTNET_RUNTIME_SOURCE}/Runtime.cs
            ${DOTNET_RUNTIME_BINDINGS}
    COMMENT "Building the engine's managed runtime")
add_custom_target(DotNet.Runtime ALL DEPENDS ${DOTNET_RUNTIME_OUT}/Engine.Runtime.dll)

# dotnet_link_compiled_runtime(<target> <name> <sources folder>): the managed runtime, its bindings
# and every .cs under <sources folder> (an IDE's bin and obj passed over), compiled ahead of time
# with NativeAOT into one static library linked into <target>, whose C# backend then calls it
# instead of starting .NET (see DotNetCompiledIn.cpp). Every type is kept through trimming, since
# behaviours are called by reflection. Linux for now.
function(dotnet_link_compiled_runtime target name sources)
    set(dir ${CMAKE_CURRENT_BINARY_DIR}/${name})
    set(out ${dir}/out)
    set(project ${dir}/Engine.Runtime.csproj)
    file(GLOB_RECURSE game_sources CONFIGURE_DEPENDS ${sources}/*.cs)
    list(FILTER game_sources EXCLUDE REGEX "/(bin|obj)/")
    file(WRITE ${project} "<Project Sdk=\"Microsoft.NET.Sdk\">
  <!-- Generated by dotnet_link_compiled_runtime: the engine's managed runtime and a game, compiled ahead of time. -->
  <PropertyGroup>
    <TargetFramework>net10.0</TargetFramework>
    <AssemblyName>Engine.Runtime</AssemblyName>
    <Nullable>enable</Nullable>
    <AllowUnsafeBlocks>true</AllowUnsafeBlocks>
    <EnableDefaultCompileItems>false</EnableDefaultCompileItems>
    <PublishAot>true</PublishAot>
    <NativeLib>Static</NativeLib>
    <RuntimeIdentifier>linux-x64</RuntimeIdentifier>
    <InvariantGlobalization>true</InvariantGlobalization>
    <NoWarn>$(NoWarn);IL2026;IL2072;IL2075;IL2070;IL2067;IL2057;IL3050</NoWarn>
  </PropertyGroup>
  <ItemGroup>
    <Compile Include=\"${DOTNET_RUNTIME_SOURCE}/Bridge.cs\" />
    <Compile Include=\"${DOTNET_RUNTIME_SOURCE}/Runtime.cs\" />
    <Compile Include=\"${DOTNET_RUNTIME_BINDINGS}\" />
    <Compile Include=\"${sources}/**/*.cs\" Exclude=\"${sources}/**/bin/**;${sources}/**/obj/**\" />
    <TrimmerRootAssembly Include=\"Engine.Runtime\" />
  </ItemGroup>
  <!-- What the program links beside the library: NativeAOT's runtime, from the package it used. -->
  <Target Name=\"WriteEngineLinkInputs\" AfterTargets=\"Publish\">
    <ItemGroup>
      <EngineLink Include=\"-Wl,--whole-archive\" />
      <EngineLink Include=\"${out}/Engine.Runtime.a\" />
      <EngineLink Include=\"-Wl,--no-whole-archive\" />
      <EngineLink Include=\"-Wl,--start-group\" />
      <EngineLink Include=\"$(IlcSdkPath)libbootstrapperdll.o\" />
      <EngineLink Include=\"$(IlcSdkPath)libRuntime.WorkstationGC.a;$(IlcSdkPath)libeventpipe-disabled.a;$(IlcSdkPath)libstandalonegc-disabled.a;$(IlcSdkPath)libRuntime.VxsortDisabled.a;$(IlcSdkPath)libaotminipal.a\" />
      <EngineLink Include=\"$(IlcFrameworkNativePath)libSystem.Native.a;$(IlcFrameworkNativePath)libstdc++compat.a;$(IlcFrameworkNativePath)libbrotlienc.a;$(IlcFrameworkNativePath)libbrotlidec.a;$(IlcFrameworkNativePath)libbrotlicommon.a;$(IlcFrameworkNativePath)libz.a\" />
      <EngineLink Include=\"-Wl,--end-group\" />
      <EngineLink Include=\"-ldl;-lm;-lrt;-lpthread\" />
    </ItemGroup>
    <WriteLinesToFile File=\"${out}/link.rsp\" Lines=\"@(EngineLink)\" Overwrite=\"true\" WriteOnlyWhenDifferent=\"true\" />
  </Target>
</Project>
")
    add_custom_command(
        OUTPUT ${out}/Engine.Runtime.a ${out}/link.rsp
        COMMAND ${DOTNET_EXECUTABLE} publish ${project} -c Release -o ${out}
                -p:BaseIntermediateOutputPath=${dir}/obj/ -nologo -v:q
        DEPENDS ${project} ${game_sources} ${DOTNET_RUNTIME_SOURCE}/Bridge.cs ${DOTNET_RUNTIME_SOURCE}/Runtime.cs
                ${DOTNET_RUNTIME_BINDINGS}
        COMMENT "Compiling the managed runtime and ${name} ahead of time")
    add_custom_target(${name}.CompiledRuntime DEPENDS ${out}/Engine.Runtime.a ${out}/link.rsp)
    add_dependencies(${name}.CompiledRuntime DotNet.Runtime) # the bindings, generated there
    add_dependencies(${target} ${name}.CompiledRuntime)
    target_link_options(${target} PRIVATE "@${out}/link.rsp")
    set_property(TARGET ${target} APPEND PROPERTY LINK_DEPENDS ${out}/Engine.Runtime.a ${out}/link.rsp)
endfunction()
